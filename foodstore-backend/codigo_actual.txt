from sqlmodel import SQLModel, Session, create_engine

DATABASE_URL = "postgresql+psycopg2://postgres:admin@localhost:5432/foodstore_db"

engine = create_engine(DATABASE_URL, echo=True)


def create_db_and_tables() -> None:
    SQLModel.metadata.create_all(engine)


def get_session():
    with Session(engine) as session:
        yield session


===== app\core\seed.py =====

from sqlmodel import Session, select

from app.core.database import engine
from app.modules.unidades_medida.model import TipoUnidad, UnidadMedida


def seed_unidades_medida() -> None:
    with Session(engine) as session:
        if session.exec(select(UnidadMedida)).first() is not None:
            return  # ya hay datos: no duplica
        session.add_all([
            UnidadMedida(nombre="kilogramo", simbolo="kg", tipo=TipoUnidad.peso),
            UnidadMedida(nombre="gramo", simbolo="g", tipo=TipoUnidad.peso),
            UnidadMedida(nombre="litro", simbolo="L", tipo=TipoUnidad.volumen),
            UnidadMedida(nombre="mililitro", simbolo="mL", tipo=TipoUnidad.volumen),
            UnidadMedida(nombre="pieza", simbolo="u", tipo=TipoUnidad.contable),
            UnidadMedida(nombre="docena", simbolo="doc", tipo=TipoUnidad.contable),
        ])
        session.commit()

    nombre: str
    descripcion: str | None = None
    activo: bool = Field(default=True)
    anulado_at: datetime | None = None

        back_populates="categorias", link_model=ProductoCategoria

===== app\modules\productos\model.py =====

from datetime import datetime
from decimal import Decimal
from typing import TYPE_CHECKING

from sqlalchemy import Column, Numeric
from sqlmodel import Field, Relationship, SQLModel

if TYPE_CHECKING:
    from app.modules.categorias.model import Categoria
    from app.modules.unidades_medida.model import UnidadMedida


class ProductoCategoria(SQLModel, table=True):
    producto_id: int = Field(foreign_key="producto.id", primary_key=True)
    categoria_id: int = Field(foreign_key="categoria.id", primary_key=True)
    es_principal: bool = Field(default=False)


class Producto(SQLModel, table=True):
    id: int | None = Field(default=None, primary_key=True)
    unidad_venta_id: int = Field(foreign_key="unidadmedida.id")
    nombre: str
    descripcion: str | None = None
    precio_base: Decimal = Field(sa_column=Column(Numeric(12, 2), nullable=False))
    stock_cantidad: Decimal = Field(
        default=Decimal("0"), sa_column=Column(Numeric(12, 3), nullable=False)
    )
    disponible: bool = Field(default=True)
    activo: bool = Field(default=True)
    anulado_at: datetime | None = None

    unidad_venta: "UnidadMedida" = Relationship(back_populates="productos")
    categorias: list["Categoria"] = Relationship(
        back_populates="productos", link_model=ProductoCategoria
    )

from enum import Enum
from typing import TYPE_CHECKING

from sqlmodel import Field, Relationship, SQLModel

if TYPE_CHECKING:
    from app.modules.productos.model import Producto


class TipoUnidad(str, Enum):
    peso = "peso"
    volumen = "volumen"
    contable = "contable"


class UnidadMedida(SQLModel, table=True):
    id: int | None = Field(default=None, primary_key=True)
    nombre: str = Field(unique=True)
    simbolo: str = Field(unique=True)
    tipo: TipoUnidad

    productos: list["Producto"] = Relationship(back_populates="unidad_venta")

from contextlib import asynccontextmanager

from fastapi import FastAPI

from app.core.database import create_db_and_tables
from app.core.seed import seed_unidades_medida
from app.modules.categorias import model as categorias_model  # noqa: F401
from app.modules.productos import model as productos_model  # noqa: F401
from app.modules.unidades_medida import model as unidades_model  # noqa: F401


@asynccontextmanager
async def lifespan(app: FastAPI):
    create_db_and_tables()
    seed_unidades_medida()
    yield


app = FastAPI(title="Food Store API", lifespan=lifespan)
